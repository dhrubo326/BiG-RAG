```python
import asyncio
import time

async def test_parallel_execution():
    """Verify asyncio.gather() works correctly."""
    print("Testing parallel execution...")

    async def task_a():
        await asyncio.sleep(0.1)
        return "A"

    async def task_b():
        await asyncio.sleep(0.1)
        return "B"

    async def task_c():
        await asyncio.sleep(0.1)
        return "C"

    # Sequential (slow)
    start = time.time()
    await task_a()
    await task_b()
    await task_c()
    sequential_time = time.time() - start
    print(f"Sequential: {sequential_time*1000:.0f}ms")

    # Parallel (fast)
    start = time.time()
    results = await asyncio.gather(task_a(), task_b(), task_c())
    parallel_time = time.time() - start
    print(f"Parallel: {parallel_time*1000:.0f}ms")

    speedup = sequential_time / parallel_time
    print(f"Speedup: {speedup:.1f}x")

    if speedup >= 2.5:
        print("✅ Parallel execution working correctly")
        return True
    else:
        print("❌ Parallel execution NOT working (speedup too low)")
        return False
```

## Architecture & Design Decisions

### Key Design Pattern: Two-Stage Path C

**Why split Path C into two stages?**

```
Stage 1 (Parallel):  Direct chunk vector search
                     ↓
                     Can run WITHOUT waiting for Path A/B
                     Uses: vdb_chunks.query()

Stage 2 (Sequential): Indirect chunk extraction
                      ↓
                      MUST wait for RRF fusion results
                      Uses: source_ids from top-5 structured knowledge
```

**Correctness Proof:**
- Direct chunks: Independent of graph results ✓
- Indirect chunks: Depend on RRF top-5 ✓
- Running Stage 1 in parallel: Safe ✓
- Running Stage 2 after RRF: Necessary ✓

### Latency Analysis (With Real Numbers)

**Sequential Execution:**
```
Path A (entities):        100ms
Path B (edges):          +100ms
Path C (chunks):          +80ms
RRF Fusion:               +10ms
Indirect chunks:          +10ms
─────────────────────────────────
Total:                    300ms
```

**Parallel Execution (Our Design):**
```
┌─ Path A:        100ms ─┐
├─ Path B:        100ms ─┤─→ max(100ms) = 100ms
└─ Path C Stage 1: 80ms ─┘
RRF Fusion:               +10ms
Indirect chunks:          +10ms
─────────────────────────────────
Total:                    120ms

Speedup: 300ms / 120ms = 2.5x faster ✓
```

**Key Insight:** Even though we add Path C, total latency only increases by 20ms (100→120ms) instead of 80ms, thanks to parallel execution.

### Error Handling Philosophy

**Fail-Safe Principle:** If any path fails, other paths should still work.

```python
# ❌ BAD: One failure breaks everything
path_a, path_b, path_c = await asyncio.gather(
    _get_node_data(...),
    _get_edge_data(...),
    _get_chunk_data(...),
)
# If path_c fails, entire query fails!

# ✅ GOOD: Isolated failure handling
try:
    path_a, path_b, path_c = await asyncio.gather(
        _get_node_data(...),
        _get_edge_data(...),
        _get_chunk_data(...),
        return_exceptions=True  # Don't fail on first exception
    )

    # Check each result
    if isinstance(path_c, Exception):
        logger.warning(f"Path C failed: {path_c}, continuing without chunks")
        path_c = []  # Empty result, not failure
except Exception as e:
    logger.error(f"Fatal error: {e}")
    raise
```

```python
async def _get_chunk_data_direct(
    query: str,
    vdb_chunks: BaseVectorStorage,
    text_chunks_db: BaseKVStorage,
    query_param: QueryParam,
) -> List[Dict]:
    """
    Stage 1 of Path C: Direct chunk vector search.

    This function CAN run in parallel with Path A and B because it has
    NO dependency on their results.

    Args:
        query: Search query string
        vdb_chunks: Vector database for chunks
        text_chunks_db: KV store for chunk metadata
        query_param: Query parameters (contains top_k)

    Returns:
        List of chunk dicts with format:
        [
            {
                "chunk_id": str,
                "content": str,
                "score": float,
                "source": "direct"
            },
            ...
        ]

    Edge Cases Handled:
        - vdb_chunks returns 0 results → return empty list
        - chunk_id not found in text_chunks_db → skip that chunk
        - vdb_chunks is None → return empty list
    """
    if vdb_chunks is None:
        logger.warning("vdb_chunks is None, skipping direct chunk search")
        return []

    try:
        # Query vector database for similar chunks
        direct_results = await vdb_chunks.query(query, top_k=5)

        if not direct_results or len(direct_results) == 0:
            logger.debug("No direct chunks found for query")
            return []

        # Fetch full chunk content from KV store
        direct_chunks = []
        for r in direct_results:
            chunk_id = r.get("id")
            if not chunk_id:
                logger.warning(f"Chunk result missing 'id' field: {r}")
                continue

            try:
                chunk_data = await text_chunks_db.get_by_id(chunk_id)
                if chunk_data and "content" in chunk_data:
                    direct_chunks.append({
                        "chunk_id": chunk_id,
                        "content": chunk_data["content"],
                        "score": float(r.get("distance", 0.0)),
                        "source": "direct"
                    })
                else:
                    logger.debug(f"Chunk {chunk_id} not found in KV store or missing content")
            except Exception as e:
                logger.warning(f"Error fetching chunk {chunk_id}: {e}")
                continue

        logger.debug(f"Retrieved {len(direct_chunks)} direct chunks")
        return direct_chunks

    except Exception as e:
        logger.error(f"Direct chunk search failed: {e}")
        # Return empty list instead of crashing
        return []
```

```python
async def _get_chunk_data_indirect(
    rrf_results: List[Dict],
    text_chunks_db: BaseKVStorage,
    max_chunks: int = 5,
) -> List[Dict]:
    """
    Stage 2 of Path C: Extract indirect chunks from RRF results.

    This function MUST run AFTER RRF fusion because it depends on
    the top-k structured knowledge results to extract source_ids.

    Args:
        rrf_results: Top-k results from RRF fusion (Path A + B)
                     Each result must have "<source_ids>" field
        text_chunks_db: KV store for chunk metadata
        max_chunks: Maximum number of indirect chunks to return

    Returns:
        List of chunk dicts (same format as direct chunks)

    Edge Cases Handled:
        - rrf_results is empty → return empty list
        - No source_ids in results → return empty list
        - Duplicate chunk IDs → deduplicated automatically (using set)
        - Chunk not found in KV store → skip that chunk
    """
    if not rrf_results or len(rrf_results) == 0:
        logger.debug("No RRF results provided for indirect chunk extraction")
        return []

    try:
        # Extract unique chunk IDs from source_ids
        indirect_chunk_ids = set()
        for result in rrf_results:
            source_ids = result.get("<source_ids>", [])
            if source_ids:
                # source_ids can be a list or a single string
                if isinstance(source_ids, list):
                    indirect_chunk_ids.update(source_ids)
                else:
                    indirect_chunk_ids.add(source_ids)

        if not indirect_chunk_ids:
            logger.debug("No source_ids found in RRF results")
            return []

        logger.debug(f"Found {len(indirect_chunk_ids)} unique indirect chunk IDs")

        # Fetch chunks from KV store
        indirect_chunks = []
        for chunk_id in list(indirect_chunk_ids)[:max_chunks]:
            try:
                chunk_data = await text_chunks_db.get_by_id(chunk_id)
                if chunk_data and "content" in chunk_data:
                    indirect_chunks.append({
                        "chunk_id": chunk_id,
                        "content": chunk_data["content"],
                        "score": 0.0,  # No direct similarity score
                        "source": "indirect"
                    })
                else:
                    logger.debug(f"Indirect chunk {chunk_id} not found or missing content")
            except Exception as e:
                logger.warning(f"Error fetching indirect chunk {chunk_id}: {e}")
                continue

        logger.debug(f"Retrieved {len(indirect_chunks)} indirect chunks")
        return indirect_chunks

    except Exception as e:
        logger.error(f"Indirect chunk extraction failed: {e}")
        return []
```

```python
async def _build_query_context(
    query: list,
    knowledge_graph_inst: BaseGraphStorage,
    vdb_entities: BaseVectorStorage,
    vdb_bipartite_edges: BaseVectorStorage,
    text_chunks_db: BaseKVStorage[TextChunkSchema],
    query_param: QueryParam,
    vdb_chunks: BaseVectorStorage = None,  # NEW parameter
    enable_reranking: bool = False,  # NEW parameter
):
    """
    Build query context using three-path retrieval with parallel execution.

    Changes from original:
    1. Added vdb_chunks parameter
    2. Added enable_reranking parameter
    3. Parallel execution of Path A, B, C using asyncio.gather()
    4. Added indirect chunk extraction
    5. Optional semantic reranking

    Returns:
        List of knowledge dicts with <knowledge>, <coherence>, <source_ids>, <type>
    """
    ll_keywords, hl_keywords = query[0], query[1]

    # === PARALLEL EXECUTION: Launch Path A, B, and C Stage 1 ===
    tasks = [
        _get_node_data(
            ll_keywords,
            knowledge_graph_inst,
            vdb_entities,
            text_chunks_db,
            query_param,
        ),
        _get_edge_data(
            hl_keywords,
            knowledge_graph_inst,
            vdb_bipartite_edges,
            text_chunks_db,
            query_param,
        ),
    ]

    # Add Path C Stage 1 (direct chunks) if vdb_chunks available
    if vdb_chunks is not None and query_param.mode in ["hybrid", "naive"]:
        tasks.append(
            _get_chunk_data_direct(
                ll_keywords,
                vdb_chunks,
                text_chunks_db,
                query_param,
            )
        )

    # Execute all paths in parallel with error handling
    try:
        if len(tasks) == 3:
            knowledge_list_1, knowledge_list_2, direct_chunks = await asyncio.gather(
                *tasks, return_exceptions=True
            )
            # Check if any task failed
            if isinstance(direct_chunks, Exception):
                logger.error(f"Path C failed: {direct_chunks}")
                direct_chunks = []
        else:
            knowledge_list_1, knowledge_list_2 = await asyncio.gather(
                *tasks, return_exceptions=True
            )
            direct_chunks = []

        # Check Path A and B results
        if isinstance(knowledge_list_1, Exception):
            logger.error(f"Path A failed: {knowledge_list_1}")
            knowledge_list_1 = []
        if isinstance(knowledge_list_2, Exception):
            logger.error(f"Path B failed: {knowledge_list_2}")
            knowledge_list_2 = []

    except Exception as e:
        logger.error(f"Parallel execution failed: {e}")
        # Fallback: return empty results
        return []

    # === RRF FUSION (Path A + B) ===
    know_score = dict()
    know_sources = dict()

    # Process Path A results
    for i, (k, source_ids) in enumerate(knowledge_list_1):
        if k not in know_score:
            know_score[k] = 0
            know_sources[k] = set()
        score = 1 / (i + 1)  # RRF formula
        know_score[k] += score
        if source_ids:
            if isinstance(source_ids, (list, set)):
                know_sources[k].update(source_ids)
            else:
                know_sources[k].add(source_ids)

    # Process Path B results
    for i, (k, source_ids) in enumerate(knowledge_list_2):
        if k not in know_score:
            know_score[k] = 0
            know_sources[k] = set()
        score = 1 / (i + 1)
        know_score[k] += score
        if source_ids:
            if isinstance(source_ids, (list, set)):
                know_sources[k].update(source_ids)
            else:
                know_sources[k].add(source_ids)

    # Sort by RRF score and get top-k
    knowledge_list = sorted(
        know_score.items(),
        key=lambda x: x[1],
        reverse=True
    )[:query_param.top_k]

    # Build structured knowledge results
    knowledge = []
    for k, score in knowledge_list:
        sources = list(know_sources.get(k, []))
        knowledge.append({
            "<knowledge>": k,
            "<coherence>": round(score, 3),
            "<source_ids>": sources,
            "<type>": "structured"
        })

    # === PATH C STAGE 2: Indirect chunks (after RRF) ===
    if vdb_chunks is not None and query_param.mode in ["hybrid", "naive"]:
        # Get indirect chunks from top-5 RRF results
        indirect_chunks = await _get_chunk_data_indirect(
            knowledge[:5],  # Top-5 structured knowledge
            text_chunks_db,
            max_chunks=5,
        )

        # Combine direct + indirect chunks
        all_chunks = direct_chunks + indirect_chunks

        # Optional semantic reranking
        if enable_reranking and len(all_chunks) > 0:
            try:
                from .reranker import _semantic_rerank
                chunk_knowledge = await _semantic_rerank(
                    ll_keywords,
                    all_chunks,
                    top_k=5
                )
                logger.debug(f"Reranked {len(all_chunks)} chunks → {len(chunk_knowledge)}")
            except ImportError:
                logger.warning("Reranker not available, returning all chunks")
                chunk_knowledge = all_chunks[:10]
            except Exception as e:
                logger.error(f"Reranking failed: {e}, using all chunks")
                chunk_knowledge = all_chunks[:10]
        else:
            # Without reranking: return all chunks (up to 10)
            chunk_knowledge = all_chunks[:10]

        # Format chunks and add to knowledge
        for chunk in chunk_knowledge:
            knowledge.append({
                "<knowledge>": chunk["content"],
                "<coherence>": round(chunk.get("score", 0.5), 3),
                "<source_ids>": [chunk["chunk_id"]],
                "<type>": "chunk"
            })

        logger.debug(f"Added {len(chunk_knowledge)} chunks to final results")

    return knowledge
```

**Scenario:** Query has no matches in any path.

**Handling:**

**Scenario:** One path takes too long and blocks others.

**Handling:**

**Handling:**

**Symptoms:**
- Query takes longer than expected
- Latency increasing over time
